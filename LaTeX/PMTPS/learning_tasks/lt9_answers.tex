\enquestion{5.55}{
    $X$ and $Y$ are independent, normal random variables with $E(X) = 2, V(X) = 5, E(Y) = 6$, and $V(Y) = 9$.
    Determine the following:

    \begin{center}
        \begin{tabular}{*{2}{m{0.45\textwidth}}}
            (a) $E(3X + 2Y)$ & (b) $V(3X + 2Y)$ \\
            (c) $P(3X + 2Y < 18)$ & (d) $P(3X + 2Y < 28)$
        \end{tabular}
    \end{center}
}
\enanswer{
    \begin{center}
        \begin{tabular}{*{2}{m{0.45\textwidth}}}
            (a) $18$ & (b) $81$ \\
            (c) $0.5$ & (d) $0.8667$
        \end{tabular}
    \end{center}
}

\enquestion{5.57}{
    A plastic casing for a magnetic disk is composed of two halves.
    The thickness of each half is normally distributed with a mean of $3$ millimeters, and a standard deviation of $0.1$ millimeter and the halves are independent.
    \begin{enumerate}[label=(\alph*)]
        \item Determine the mean and standard deviation of the total thickness of the two halves.
        \item What is the probability that the total thickness exceeds 6.3 millimeters?
    \end{enumerate}
}
\enanswer{
    \begin{enumerate}[label=(\alph*)]
        \item $\mu_{\text{sum}} = 6, \quad \sigma_{\text{sum}} = 0.1414$
        \item $0.0169$
    \end{enumerate}
}

\enquestion{5.60}{
    The width of a casing for a door is normally distributed with a mean of $61$ cm and a standard deviation of $0.4$ cm.
    The width of a door is normally distributed with a mean of $60.5$ cm and a standard deviation of $0.2$ cm.
    Assume independence.
    \begin{enumerate}[label=(\alph*)]
        \item Determine the mean and standard deviation of the difference between the width of the casing and the width of the door.
        \item What is the probability that the width of the casing minus the width of the door exceeds $0.8$ cm?
    \end{enumerate}
}
\enanswer{
    \begin{enumerate}[label=(\alph*)]
        \item $\mu_{\text{diff}} = 0.5, \quad \sigma_{\text{diff}} = 0.4472$
        \item $0.2512$
    \end{enumerate}
}

\enquestion{5.63}{
    The photoresist thickness in semiconductor manufacturing has a mean of $10$ micrometers and a standard deviation of $1$ micrometer.
    Assume that the thickness is normally distributed and that the thicknesses of different wafers are independent.
    \begin{enumerate}[label=(\alph*)]
        \item Determine the probability that the average thickness of $10$ wafers is either greater than $12$ or less than $8$ micrometers.
        \item Determine the number of wafers that need to be measured such that the probability that the average thickness exceeds $11$ micrometers is $0.05$.
        \item If the mean thickness is $10$ micrometers, what should the standard deviation of thickness equal so that the probability that the average of $10$ wafers is either more than $11$ or less than $9$ micrometers is $0.001$?
    \end{enumerate}
}
\enanswer{
    \begin{enumerate}[label=(\alph*)]
        \item $0.0000$
        \item $3$
        \item $0.9610$
    \end{enumerate}
}

\enquestion{5.64}{
    Assume that the weights of individuals are independent and normally distributed with a mean of $75\text{ kg}$ and a standard deviation of $15\text{ kg}$.
    Suppose that $25$ people squeeze into an elevator that is designed to hold $1950\text{ kg}$.
    \begin{enumerate}[label=(\alph*)]
        \item What is the probability that the load (total weight) exceeds the design limit?
        \item What design limit is exceeded by $25$ occupants with probability $0.0002$?
    \end{enumerate}
}
\enanswer{
    \begin{enumerate}[label=(\alph*)]
        \item $0.1587$
        \item $2140.5063$
    \end{enumerate}
}

\enquestion{5.90}{
    Suppose that $f_{XYZ}(x,y,z)$ is constant over the region $x^2 + y^2 \le 1, 0 < z < 4$.
    Determine the following:

    \begin{tabular}{*{2}{m{0.45\textwidth}}}
        (a) $P(X^{2} + Y^{2} \le 0.5)$ & (b) $P(X^{2} + Y^{2} \le 0.5, Z < 1)$ \\
        \multicolumn{2}{l}{(c) Joint conditional probability density function of $X$ and $Y$ given that $Z = 1$}\\
        \multicolumn{2}{l}{(d) Marginal probability density function of $X$}\\
        \multicolumn{2}{l}{(e) Conditional mean of $Z$ given that $X = 0$ and $Y = 0$}\\
        \multicolumn{2}{l}{(f) Conditional mean of $Z$ given that $X = x$ and $Y = y$}
    \end{tabular}
}
\enanswer{
    \begin{tabular}{*{2}{m{0.45\textwidth}}}
        (a) $0.5$ & (b) $0.125$ \\
        \multicolumn{2}{l}{(c) $f_{XY|1}(x,y) = 1/\pi$ for $x^2 + y^2 \le 1$} \\
        \multicolumn{2}{l}{(d) $f_X(x) = \frac{2}{\pi}\sqrt{1-x^2}$ for $-1 < x < 1$} \\
        \multicolumn{2}{l}{(e) Since $f_{Z|0,0}(z) = 0.25$ for $0 < z < 4$, $\mu_{Z|0,0} = 2$} \\
        \multicolumn{2}{l}{(f) Since $f_{Z|x,y}(z) = 0.25$ for $0 < z < 4$, $E(Z|(X,Y) = (x,y)) = 2$}
    \end{tabular}
}
