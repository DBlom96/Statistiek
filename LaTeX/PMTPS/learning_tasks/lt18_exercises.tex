\enquestion{9.96}{
    Consider the following frequency table of observations on the random variable $X$.
    \begin{center}
        \begin{tabular}{l*{5}{c}}
            \toprule
            Values & $0$ & $1$ & $2$ & $3$ & $4$ \\
            Observed frequency & $24$ & $30$ & $31$ & $11$ & $4$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{enumerate}[label=(\alph*)]
        \item Based on these $100$ observations, is a Poisson distribution with a mean of $1.5$ an appropriate model?
        Perform a goodness-of-fit procedure with $\alpha = 0.05$.
        \item Calculate the $p$-value for this test.
    \end{enumerate}
}

            \bottomrule
        \end{tabular}
    \end{center}
    \begin{enumerate}[label=(\alph*)]
