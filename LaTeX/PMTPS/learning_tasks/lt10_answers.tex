\enquestion{7.3}{
    PVC pipe is manufactured with a mean diameter of $\qty{2.565}{\cm}$ and a standard deviation of $\qty{0.008}{\cm}$.
    Find the probability that a random sample of $n = 5$ sections of pipe will have a sample mean diameter greater than $\qty{2.560}{\cm}$ and less than $\qty{2.570}{\cm}$.
}
\enanswer{
    Central Limit Theorem: $\bar{X} \sim N(\mu = 2.565, \sigma = 0.008 / \sqrt{5} \approx 0.0036)$ \\
    $0.8377$
}

\enquestion{7.7}{
    The compressive strength of concrete (in \si{\kN\per\metre\squared}) is normally distributed with $\mu = 17237$ and $\sigma = 345$.
    Find the probability that a random sample of $n = 5$ specimens will have a sample mean compressive strength that falls in the interval $\qty{17230}{\kN\per\metre\squared}$ to $\qty{17305}{\kN\per\metre\squared}$.
}
\enanswer{
    Central Limit Theorem: $\bar{X} \sim N(\mu = 17237, \sigma = 345 / \sqrt{5} \approx 154.2887)$ \\
    $0.1884$
}

\enquestion{7.10}{
    Suppose that the random variable $X$ has the continuous uniform distribution
    \[
        f(x) = \begin{cases}
            1, & 0 \le x \le 1 \\
            0, & \text{otherwise}
        \end{cases}
    \]
    Suppose that a random sample of $n = 27$ observations is selected from this distribution.
    What is the approximate probability distribution of $\bar{X} + 1 / 2$?
    Find the mean and variance of this quantity.
}
\enanswer{
    Central Limit Theorem: $\bar{X} + 1/2 \sim N\left(\mu = 1, \sigma = 1/18 \right)$.
    The mean is equal to $1$ and the variance equal to $\left(\frac{1}{18} \right)^2 = \frac{1}{324}$.
}

\enquestion{7.11}{
    Suppose that $X$ has a discrete uniform distribution
    \[
        f(x) = \begin{cases}
            1/4, & x = 1, 2, 3, 4 \\
            0, & \text{otherwise}
        \end{cases}
    \]
    A random sample of $n = 36$ is selected from this population.
    Find the probability that the sample mean is greater than $2.3$ but less than $2.7$, assuming that the sample mean would be measured to the nearest tenth.
}
\enanswer{
    Central Limit Theorem: $\bar{X} \sim N(\mu = 2.5, \sigma = \frac{1}{12}\sqrt{5})$ \\
    Correction for continuity, without rounding to one decimal: $P(2.3 < \bar{X} < 2.7) = 0.7169$ \\
    Correction for continuity, rounded to one decimal (notice that we need to have unrounded values between $2.35$ and $2.65$, 
    to get rounded values of $2.4$, $2.5$ or $2.6$): $P(2.35 < \bar{X} < 2.65) = 0.5792$
}

\enquestion{7.12}{
    The amount of time that a customer spends waiting at an airport check-in counter is a random variable with mean $\qty{8.2}{\minute}$ and standard deviation $\qty{6}{\minute}$.
    Suppose that a random sample of $n = 36$ customers is observed.
    Find the probability that the average time waiting in line for these customers is

    \begin{tabular}{*{2}{m{0.45\textwidth}}}
        \multicolumn{2}{l}{(a) Less than $\qty{10}{\minute}$} \\
        \multicolumn{2}{l}{(b) Between $\qty{5}{\minute}$ and $\qty{10}{\minute}$} \\
        \multicolumn{2}{l}{(c) Less than $\qty{6}{\minute}$}
    \end{tabular}
}
\enanswer{
    Central Limit Theorem: $\bar{X} \sim N(\mu = 8.2, \sigma = 6/\sqrt{36} = 1)$
    
    \begin{tabular}{*{3}{m{0.3\textwidth}}}
        (a) $0.9641$ & (b) $0.9634$ & (c) $0.0139$
    \end{tabular}
}

\enquestion{7.13}{
    A random sample of size $n_{1} = 16$ is selected from a normal population with a mean of $75$ and a standard deviation of $8$.
    A second random sample of size $n_{2} = 9$ is taken from another normal population with mean $70$ and standard deviation $12$.
    Let $\bar{X}_{1}$ and $\bar{X}_{2}$ be the two sample means.
    Find:
    \begin{enumerate}[label=(\alph*)]
        \item The probability that $\bar{X}_{1} - \bar{X}_{2}$ exceeds $4$
        \item The probability that $2.5 \le \bar{X}_{1} - \bar{X}_{2} \le 5.5$
    \end{enumerate}
}
\enanswer{
    Central Limit Theorem: $\bar{X}_1 \sim N(\mu=75, \sigma = 8 / \sqrt{16} = 2)$ and $\bar{X}_2 \sim N(\mu=70, \sigma = 12 / \sqrt{9} = 4)$.
    Therefore: $D = \bar{X}_{1} - \bar{X}_{2} \sim N(\mu = 75-70=5, \sigma = \sqrt{2^2+4^2}=\sqrt{20})$.
    \begin{enumerate}[label=(\alph*)]
        \item $0.5885$
        \item $0.2564$
    \end{enumerate}
}
