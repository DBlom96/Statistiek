\enquestion{9.96}{
    Consider the following frequency table of observations on the random variable $X$.
    \begin{center}
        \begin{tabular}{l*{5}{c}}
            \toprule
            Values & $0$ & $1$ & $2$ & $3$ & $4$ \\
            Observed frequency & $24$ & $30$ & $31$ & $11$ & $4$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    \begin{enumerate}[label=(\alph*)]
        \item Based on these $100$ observations, is a Poisson distribution with a mean of $1.5$ an appropriate model?
        Perform a goodness-of-fit procedure with $\alpha = 0.05$.
        \item Calculate the $p$-value for this test.
    \end{enumerate}
}
\enanswer{
    \begin{enumerate}
        \item Notice that in this exercise, the value of the Poisson parameter $\mu = 1.5$ is given, so we do not need to estimate it.
        The hypotheses in this test are as follows:
        \begin{align*}
            &H_0: \text{$X$ follows a Poisson distribution with mean $\mu = 1.5$.} \\
            &H_1: \text{$X$ does not follow a Poisson distribution with mean $\mu = 1.5$.}
        \end{align*}
        We calculate the expected frequencies of the value $0$, $1$, $2$, $3$ and $4$ using the $\poissonpdf$ function:

        \begin{tabular}{ccc}
            \toprule
                \textbf{Categorie ($i$)} & \textbf{Observed ($O_i$)} & \textbf{Expected ($E_i$)} \\
            \midrule
                0        & $24$ & $100\cdot\poissonpdf(\mu=1.5, k=0)\approx22.3130$ \\
                1        & $30$ & $100\cdot\poissonpdf(\mu=1.5, k=1)\approx33.4695$ \\
                2        & $31$ & $100\cdot\poissonpdf(\mu=1.5, k=2)\approx25.1021$ \\
                3        & $11$ & $100\cdot\poissonpdf(\mu=1.5, k=3)\approx12.5511$ \\
                $\geq 4$ & $4$  & $100\cdot(1-\poissoncdf(\mu=1.5, k=3))\approx6.5642$ \\
            \bottomrule
        \end{tabular}

        The $\chi^2$ test statistic can then be computed as follows:
        \begin{align*}
            \chi^2 &= \frac{(O_{0} - E_{0})^2}{E_{0}} + \frac{(O_{1} - E_{1})^2}{E_{1}} + \frac{(O_{2} - E_{2})^2}{E_{2}} + \frac{(O_{3} - E_{3})^2}{E_{3}} + \frac{(O_{\geq 4} - E_{\geq 4})^2}{E_{\geq 4}}\\
                    &= \frac{(24 - 22.3130)^2}{22.3130} + \ldots + \frac{(4 - 6.5642)^2}{6.5642}\\
                    &\approx 3.0663.
        \end{align*}
        Under assumption of $H_0$, the test statistic comes from a chi-square distribution with $\df = n - p - 1 = 5 - 0 - 1 = 4$ degrees of freedom,
        where $n$ is the number of categories and $p$ the number of estimated parameters.
        The critical region ($\alpha = 0.05$, $\chi^2$-test) is $Z = [C_R, \infty)$, and we can determine the value of $C_R$
        using the \enquote{numeric solver}:
        \begin{align*}
            y_1 &= \chi^2\text{cdf}(\lb=X, \ub=\GRinfinity, \df=5-0-1 = 4), \\
            y_2 &= \alpha = 0.05.
        \end{align*}
        The numeric solver returns a value $C_R \approx 9.4877$.
        As $\chi^2 = 3.0663$ does not lie in the critical region, we fail to reject $H_0$.

        \item The $p$-value corresponding to $\chi^2 = 3.0663$ is 
        \begin{align*}
            p = P(X^2 \geq 3.0663)  &= \chi^2\text{cdf}(\lb=3.0663, \ub=\GRinfinity, \df=n-p-1 = 4) \\
                                    &\approx 0.5468.
        \end{align*}
        As $p > \alpha = 0.05$, we again fail to reject $H_0$.
        There is not enough evidence to reject the claim that the data comes from a Poisson distribution with mean $\mu=1.5$.
    \end{enumerate}
}

                8           & $9$  &  $75 \cdot (1 - \poissoncdf(\mu=368/75, k=7)) \approx 9.8403$ \\
            \bottomrule
        \end{tabular}
        We need to combine the categories $\leq 1$ and $2$ to get an observed frequency of $1+11=12$ and 
        an expected frequency of $3.2769 + 6.6783 = 9.8665$. We now have seven categories left:

        \begin{tabular}{ccc}
            \toprule
                \textbf{Categorie ($i$)} & \textbf{Observed ($O_i$)} & \textbf{Expected ($E_i$)} \
            \midrule
                $\leq 2$ & $12$ & $9.9552$ \\
                3 & $8$  & $10.9227$ \\
                4 & $13$ & $13.3986$ \\
                5 & $11$ & $13.1484$ \\
                6 & $12$ & $10.7525$ \\
                7 & $10$ & $7.5370$ \\
                8 & $9$  & $12.5624$ \\

        \begin{align*}
            y_1 &= \chi^2\text{cdf}(\lb=X, \ub=\GRinfinity, \df=5), \\
            y_2 &= \alpha = 0.05.
        \end{align*}
        The numeric solver returns a value $C_R \approx 11.0705$.
        As $\chi^2 = 2.5234$ does not lie in the critical region, we fail to reject $H_0$.
        
        \item The $p$-value corresponding to $\chi^2 = 2.5234$ is
        \begin{align*}
            p = P(X^2 \geq 2.5234) &= \chi^2\text{cdf}(\lb=2.5234, \ub=\GRinfinity, \df=5)\\
                                    &\approx 0.7730
        \end{align*}
        \begin{align*}
            p = P(\chi^2 > 2.5234) &= \chi^2\text{cdf}(\text{lower}=2.5234; \text{upper}=10^{99}; \text{df}=5) \\
                                             &\approx 0.7730.   
        \end{align*}
        As $p > \alpha = 0.05$, we again fail to reject $H_0$.
        There is not enough evidence to reject the claim that the data comes from a Poisson distribution with $\mu$.
    \end{enumerate}
}

            \bottomrule
        \end{tabular}

        We only have four categories left:
        \begin{tabular}{ccc}
            \toprule
                \textbf{Categorie ($i$)} & \textbf{Observed ($O_i$)} & \textbf{Expected ($E_i$)} \\
            \midrule
                0        & $4$  & $100\cdot\binompdf(n=6, p=0.25, k=0)\approx 8.8989$ \\
                1        & $14$ & $100\cdot\binompdf(n=6, p=0.25, k=1)\approx 17.7979$ \\

            y_2 &= \alpha = 0.05.
        \end{align*}
        The numeric solver returns a value $C_R \approx 7.8147$.
        As $\chi^2 = 26.6844$ lies in the critical region, we reject $H_0$.

        \item The $p$-value corresponding to $\chi^2 = 26.6844$ is 
        \begin{align*}
            p = P(X^2 \geq 26.6844)  &= \chi^2\text{cdf}(\lb=26.6844, \ub=\GRinfinity, \df=4-0-1 = 3) \\
                                     &\approx 0.0000.
        \end{align*}
        As $p < \alpha = 0.05$, we again reject $H_0$.

    \end{enumerate}
