\enquestion{5.1}{
    Show that the following function satisfies the properties of a joint probability mass function.

    \begin{center}
        \begin{tabular}{*{3}{c}}
            \toprule
                $x$     & $y$   & $f_{XY}(x,y)$ \\
            \midrule
                $1$     & $1$   & $1/4$ \\
                $1.5$   & $2$   & $1/8$ \\
                $1.5$   & $3$   & $1/4$ \\
                $2.5$   & $4$   & $1/4$ \\
                $3$     & $5$   & $1/8$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    
    Determine the following:
    
    \begin{tabular}{*{2}{C{.45\textwidth}}}
        (a) $P(X<2.5, Y<3)$ & (b) $P(X<2.5)$ \\
        (c) $P(Y<3)$        & (d) $P(X>1.8, Y>4.7)$ \\
        \multicolumn{2}{l}{(e) $E(X)$, $E(Y)$, $V(X)$, $V(Y)$} \\
    \end{tabular}
}

\enquestion{5.5}{
    In the transmission of digital information, the probability that a bit has high, moderate and low distortion is $0.01$, $0.04$, and $0.95$, respectively.
    Suppose that three bits are transmitted and that the amount of distortion of each bit is assumed to be independent.
    Let $X$ and $Y$ denote the number of bits with high and moderate distortion out of the three, respectively.
    Determine:

    \begin{tabular}{*{2}{C{.45\textwidth}}}
        (a) $f_{XY}(x,y)$   & (b) $f_X(x)$ \\
        (c) $E(Y)$          & (d) $f_{Y|2}(y)$ \\
        (e) $E(Y|X=1)$      & (f) Are $X$ and $Y$ independent? \\
    \end{tabular}
}
