\begin{frame}{Axioma's van de kansrekening}
    Stel nu dat we een kansexperiment hebben met een uitkomstenruimte $S$, en een gebeurtenis $E \subseteq S$ ($E$ is een \enquote{deelverzameling} van de uitkomstenruimte).
    \vfill
    Een \textbf{\color{concept} kansfunctie} $P$ is een functie die aan iedere gebeurtenis $E$ een getal toekent zodat het volgende geldt:
    \begin{itemize}[<+->]
        \item $P(S) = 1$
        \item $0 \leq P(E) \leq 1$ voor iedere gebeurtenis $E$.
        \item Als twee gebeurtenissen $E_1$ en $E_2$ geen overlap hebben ($E_1 \cap E_2 = \emptyset$), dan geldt $P(E_1 \cup E_2) = P(E_1) + P(E_2)$.
    \end{itemize}
\end{frame}

\begin{frame}{Handige rekenregels voor kansrekening}
    Met deze axioma's kunnen we een aantal handige rekenregels afleiden:
    \begin{itemize}\setlength\itemsep{4em}
        \item Voor een gebeurtenis $E$ geldt voor iedere uitkomst $i$ dat $i$ onderdeel is van $E$ of van $\bar{E}$, oftewel $P(S) = P(E \cup \bar{E}) = P(E) + P(\bar{E}) = 1$:
        \bigskip
       
        $P(E) = 1 - P(\bar{E})$ \hfill (\emph{complementregel})

        \item Als we twee gebeurtenissen hebben $E_1$ en $E_2$ met een overlap, dan tellen we de overlap dubbel als we $P(E_1)$ en $P(E_2)$ optellen.
        Met andere woorden: \bigskip
        
        $P(E_1 \cup E_2) = P(E_1) + P(E_2) - P(E_1 \cap E_2)$ \hfill (\emph{algemene optelregel})
    \end{itemize}
\end{frame}

    \only<+>{
        \textbf{Wat is de kans dat gebeurtenis $B$ niet plaatsvindt?}\bigskip

        $B$ bestaat uit $6$ uitkomsten, dus $P(B) = \frac{6}{36}$.
        Met de complementregel vinden we dan 
        \begin{align*}
            P(\bar{B})  &= 1 - P(B) \\
                        &= 1 - \frac{6}{36} \\
                        &= \frac{30}{36} = \frac{5}{6}.
        \end{align*}
    }

    \only<+>{
        \textbf{Wat is de kans dat minstens één van beide gebeurtenissen $A$ of $B$ plaatsvindt?}\bigskip

        $A$ bestaat uit $11$ uitkomsten, $B$ bestaat uit $6$ uitkomsten, en $A \cap B$ bestaat uit $2$ uitkomsten, oftewel 
        \begin{itemize}
            \item $P(A) = \frac{11}{36}$
            \item $P(B) = \frac{6}{36}$
            \item $P(A \cap B) = \frac{2}{36}$
        \end{itemize}
        Met de algemene optelregel vinden we dan 
        \begin{align*}
            P(A \cup B) &= P(A) + P(B) - P(A \cap B) \\
                        &= \frac{11}{36} + \frac{6}{36} - \frac{2}{36} \\
                        &= \frac{15}{36}.
        \end{align*}
    }

    \only<+>{
        \textbf{Wat is de kans dat zowel $A$ als $C$ plaatsvindt?}\bigskip

        Deze kans is gelijk aan $0$. 
        
        Gebeurtenis $C$ vindt alleen plaats als er een $2$ en een $5$ worden gegooid. 
        
        In dat geval is er geen vier gegooid, dus kan $A$ niet tegelijkertijd met $C$ plaatsvinden.
    }
\end{frame}

            \draw (C.east) -- (stop);
        \end{tikzpicture}
    \end{center}\pause
    \vfill
    Introduceer gebeurtenissen $A, B, C$ dat de respectievelijke componenten blijven functioneren.
    We willen dan $P\left((A\cup B) \cap C\right)$ berekenen:\pause
    \begin{itemize}[<+->]
        \item $P(A) = 0.95$, $P(B) = 0.88$ en $P(C) = 0.92$.
        \item $P(A \cup B) = P(A) + P(B) - P(A \cap B) = P(A) + P(B) - P(A)\cdot P(B) =  0.95 + 0.88 - 0.95 \cdot 0.88 = 0.994$.
        \item $P((A\cup B) \cap C) = P(A \cup B) \cdot P(C) = 0.994 \cdot 0.92 = 0.91448$.
    \end{itemize}\pause
    \textbf{De kans dat het circuit blijft functioneren is dus $0.91448$.}
\end{frame}
